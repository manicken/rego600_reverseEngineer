before it was using:
@0x7B UART_RX_BYTE_TEMP
@0x7C CMD_BYTE
@0x7D CHECKSUM byte
@0x7E state
@0x7F payload byte index
@0x80 RX_DONE_FLAG

now i think more slim,
use bank switch of R0-R7 and uses bank3 entirely for UART ISR:
R0 UART_RX_BYTE_PTR (range 0x80-0x86)
R1 UART_TX_BYTE_PTR (future use maybe if we can use IRAM as TX buffer
R7 UART_RX_BYTE_TEMP
@0x7B UART_RX_DONE_FLAG
@0x80 UART_RX_BUFF_START (it also now stores CMD_BYTE)


;UART new ISR handler direct in vector area,
;uses unused area before UART vector that normally belong to 
;TIMER1 overflow ISR, but as timer2 is not used this is good

ORG 0x001D
ORG_END 0x0022

UART_RX_IRQ_HANDLER_TRAMPOLINE:
    LJMP UART_RX_IRQ_HANDLER

UART_TX_IRQ_HANDLER_TRAMPOLINE:
    RETI
    
ORG 0x0023
ORG_END 0x002A

UART_IRQ_VECTOR:
    JBC RI,UART_RX_IRQ_HANDLER_TRAMPOLINE
    JBC TI,UART_TX_IRQ_HANDLER_TRAMPOLINE
    RETI


ORG 0x6A6A
UART_RX_IRQ_HANDLER:
	PUSH A
	PUSH PSW

uart_rx_check_if_uart_done_is_set:
	MOV A,R2 ; use R2 as RX_DONE_FLAG by then cheking in main loop from direct adress 0x1A
	NOP
	JNZ uart_rx_irq_skip
MOV PSW,#0x18

uart_rx_one_byte:
	MOV A,S0BUF
	MOV R7,A ;(1 cycle) uses R7 as current RX byte instead of @7B

uart_rx_check_start_byte:
	RLC A
	JNC uart_rx_bytes_start_byte_rx_check
	XRL A,#0x02 ; 0x81 rotated left with C=0 -> 0x02
	JNZ uart_rx_reset

uart_rx_start_received:
	MOV R0,#0x80
	MOV R6,#0x01 ; use R6 as start_byte_rx_flag
	SJMP uart_rx_irq_end

uart_rx_bytes_start_byte_rx_check:
	MOV A,R6
	JNZ uart_rx_irq_end ; jump if start byte have not been rx

uart_rx_bytes:
	MOV A,R7
	MOV @R0,A
	INC R0
	CJNE R0,#0x88,uart_rx_irq_end ; here we have rx all 8 bytes

uart_rx_done:
	MOV R2,#0x01 ; use R2 as RX_DONE_FLAG by then cheking in main loop from direct adress 0x1A

uart_rx_reset:
	MOV R0,#0x00
	SJMP uart_rx_irq_end

uart_rx_irq_skip: 
	MOV A,S0BUF  ; allways consume rx bytes

uart_rx_irq_end:
	POP PSW
	POP A
	RETI
